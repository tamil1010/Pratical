trigger PreventAccountDeletion on Account (before delete) {
    Set<Id> accountIds = Trigger.oldMap.keySet();
    List<Opportunity> openOpportunities = [
        SELECT Id, AccountId, StageName
        FROM Opportunity
        WHERE AccountId IN :accountIds
        AND StageName NOT IN ('Closed Won', 'Closed Lost')
    ];
    Set<Id> accountIdsWithOpenOpp = new Set<Id>();
    for (Opportunity opp : openOpportunities) {
        accountIdsWithOpenOpp.add(opp.AccountId);
    }
    for (Account acc : Trigger.old) {
        if (accountIdsWithOpenOpp.contains(acc.Id)) {
            acc.addError(
                'Cannot delete this Account because it has Open Opportunities.'
            );
        }
    }
}

public class UpdateContactMailingCity implements Queueable {

    private Set<Id> accountIds;

    public UpdateContactMailingCity(Set<Id> ids) {
        accountIds = ids;
    }

    public void execute(QueueableContext context) {

        List<Contact> contacts = [
            SELECT Id, MailingCity, Account.BillingCity
            FROM Contact
            WHERE AccountId IN :accountIds
        ];

        for (Contact c : contacts) {
            c.MailingCity = c.Account.BillingCity;
        }

        if (!contacts.isEmpty()) {
            update contacts;
        }
    }
}

trigger AccountBillingCityTrigger on Account (after update) {

    Set<Id> accountIds = new Set<Id>();

    for (Account acc : Trigger.new) {

        Account oldAcc = Trigger.oldMap.get(acc.Id);

        if (acc.BillingCity != oldAcc.BillingCity) {
            accountIds.add(acc.Id);
        }
    }

    if (!accountIds.isEmpty()) {
        System.enqueueJob(
            new UpdateContactMailingCity(accountIds)
        );
    }
}

trigger PreventDuplicateContacts on Contact (before insert) {

    Set<String> emails = new Set<String>();
    Set<String> phones = new Set<String>();

    for (Contact c : Trigger.new) {
        if (c.Email != null) emails.add(c.Email.toLowerCase());
        if (c.Phone != null) phones.add(c.Phone);
    }

    List<Contact> existingContacts = [
        SELECT Email, Phone
        FROM Contact
        WHERE Email IN :emails OR Phone IN :phones
    ];

    Set<String> existingEmails = new Set<String>();
    Set<String> existingPhones = new Set<String>();

    for (Contact c : existingContacts) {
        if (c.Email != null) existingEmails.add(c.Email.toLowerCase());
        if (c.Phone != null) existingPhones.add(c.Phone);
    }

    for (Contact c : Trigger.new) {
        if (c.Email != null && existingEmails.contains(c.Email.toLowerCase())) {
            c.addError('Duplicate Contact Email!');
        }

        if (c.Phone != null && existingPhones.contains(c.Phone)) {
            c.addError('Duplicate Contact Phone!');
        }
    }
}

global class DailyOpportunitySummaryBatch
implements Database.Batchable<SObject>, Database.Stateful, Schedulable {

    global List<String> summaries = new List<String>();

    global Database.QueryLocator start(Database.BatchableContext bc) {
        return Database.getQueryLocator([
            SELECT Name, StageName, Amount, CloseDate, Owner.Name
            FROM Opportunity
            WHERE CreatedDate = TODAY
        ]);
    }

    global void execute(Database.BatchableContext bc, List<Opportunity> scope) {
        for (Opportunity opp : scope) {
            summaries.add(
                opp.Name + ' | ' + opp.StageName +
                ' | Amount: ' + opp.Amount +
                ' | Owner: ' + opp.Owner.Name
            );
        }
    }

    global void finish(Database.BatchableContext bc) {
        User manager = [
            SELECT Name, Email
            FROM User
            WHERE UserRole.Name = 'Sales Manager'
            AND IsActive = true
            LIMIT 1
        ];

        Messaging.SingleEmailMessage mail = new Messaging.SingleEmailMessage();
        mail.setToAddresses(new String[]{manager.Email});
        mail.setSubject('Daily Opportunity Summary');

        String body = 'Daily Opportunity Summary\n\n';

        for (String s : summaries) {
            body += s + '\n';
        }

        if (summaries.isEmpty()) {
            body += 'No opportunities created today.';
        }

        mail.setPlainTextBody(body);
        Messaging.sendEmail(new Messaging.SingleEmailMessage[]{mail});
    }

    global void execute(SchedulableContext sc) {
        Database.executeBatch(new DailyOpportunitySummaryBatch(), 200);
    }
}

public class OpportunitySharingHelper {
    public static void shareRecords(List<Opportunity> opps) {
        Set<Id> accIds = new Set<Id>();
        for (Opportunity o : opps)
            if (o.AccountId != null) accIds.add(o.AccountId);
        List<OpportunityShare> shares = new List<OpportunityShare>();
        for (AccountTeamMember m : [
            SELECT AccountId, UserId, OpportunityAccessLevel
            FROM AccountTeamMember
            WHERE AccountId IN :accIds
        ]) {
            for (Opportunity o : opps) {
                if (o.AccountId == m.AccountId &&
                    m.OpportunityAccessLevel != 'None' &&
                    o.OwnerId != m.UserId) {
                    shares.add(new OpportunityShare(
                        OpportunityId = o.Id,
                        UserOrGroupId = m.UserId,
                        OpportunityAccessLevel = m.OpportunityAccessLevel,
                        RowCause = Schema.OpportunityShare.RowCause.Manual
                    ));
                }
            }
        }
        if (!shares.isEmpty())
            Database.insert(shares, false);
    }
}

trigger InventoryUpdateTrigger on Order_Item__c (after insert, after delete) {

    Map<Id, Decimal> quantities = new Map<Id, Decimal>();

    if (Trigger.isInsert) {
        for (Order_Item__c item : Trigger.new) {
            quantities.put(
                item.Product_Inventory__c,
                (quantities.containsKey(item.Product_Inventory__c)
                    ? quantities.get(item.Product_Inventory__c) : 0)
                - item.Quantity__c
            );
        }
    }

    if (Trigger.isDelete) {
        for (Order_Item__c item : Trigger.old) {
            quantities.put(
                item.Product_Inventory__c,
                (quantities.containsKey(item.Product_Inventory__c)
                    ? quantities.get(item.Product_Inventory__c) : 0)
                + item.Quantity__c
            );
        }
    }

    List<Product_Inventory__c> stocks = [
        SELECT Id, Available_Quantity__c
        FROM Product_Inventory__c
        WHERE Id IN :quantities.keySet()
    ];

    for (Product_Inventory__c stock : stocks) {
        stock.Available_Quantity__c =
            stock.Available_Quantity__c + quantities.get(stock.Id);
    }

    update stocks;
}

public with sharing class LeadSourceController {
    public class LeadSourceData {
        @AuraEnabled
        public String leadSource;
        @AuraEnabled
        public Integer leadCount;
        public LeadSourceData(String source, Integer count) {
            leadSource = source;
            leadCount = count;
        }
    }
    @AuraEnabled(cacheable=true)
    public static List<LeadSourceData> getOpenLeadsBySource() {
        List<LeadSourceData> results =
            new List<LeadSourceData>();
        List<AggregateResult> groupedLeads = [
            SELECT LeadSource, COUNT(Id) total
            FROM Lead
            WHERE IsConverted = false
            AND Status != 'Closed - Not Converted'
            GROUP BY LeadSource
        ];
        for (AggregateResult ar : groupedLeads) {
            String source = (String) ar.get('LeadSource');
            if (String.isBlank(source)) {
                source = 'Other';
            }
            Integer count = (Integer) ar.get('total');
            results.add(new LeadSourceData(source, count));
        }
        return results;
    }
}
