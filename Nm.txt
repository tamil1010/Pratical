##1

1. Log in to Salesforce and click the Setup gear icon.
2.    In Quick Find, search for Flows. Open Flows and click New Flow.
3.    Select Record-Triggered Flow and click Create.
4.    Select Opportunity as the Object.
5.    Under Configure Trigger, select A record is updated.
6.    Under Entry Conditions, set Field = Stage, Operator = Equals, and Value = Closed Won.
7.    For When to Run the Flow for Updated Records, select Only when a record is updated to meet the condition requirements. This makes the Flow run only when Stage changes to Closed Won.
8.    Under Optimize the Flow For, select Actions and Related Records, then click Done.
9.    On the Flow canvas, click the plus (+) below Start and select Get Records.
10.  Set Label = Get Related Account and Object = Account.
11.  Set the Get Records condition: Field = Id, Operator = Equals, Value = $Record.AccountId.
12.  Choose Only the first record to be stored and click Done.
13.  Click the plus (+) after Get Records and select Decision.
14.  Set Label = Check Account Found. Create an outcome named Account Found.
15.  For the Account Found condition, select Get Related Account > Id, Operator = Is Null, Value = False. This checks that an Account was found.
16.  Connect the Account Found outcome to an Update Records element. Leave the Default Outcome connected to End so Opportunities without an Account are skipped.
17.  In Update Records, set Label = Update Account Type.
18.  Choose Specify conditions to identify records, and set fields individually. Select Object = Account.
19.  Set the condition: Field = Id, Operator = Equals, Value = Get Related Account > Id.
20.  Under Set Field Values, set Field = Type and Value = Customer - Direct. Click Done.
21.  Check the flow path: Start → Get Related Account → Check Account Found → (Account Found) Update Account Type → End. The Default Outcome should go directly to End.
22.  Click Save. Enter Flow Label = Update Account Type When Opportunity Is Won, then click Save.
23.  Click Activate to activate the Flow.

Name: PreventAccountDeletion
sObject: Account

trigger PreventAccountDeletion on Account (before delete) {
    Set<Id> accountIds = Trigger.oldMap.keySet();
    List<Opportunity> openOpportunities = [
        SELECT Id, AccountId, StageName
        FROM Opportunity
        WHERE AccountId IN :accountIds
        AND StageName NOT IN ('Closed Won', 'Closed Lost')
    ];
    Set<Id> accountIdsWithOpenOpp = new Set<Id>();
    for (Opportunity opp : openOpportunities) {
        accountIdsWithOpenOpp.add(opp.AccountId);
    }
    for (Account acc : Trigger.old) {
        if (accountIdsWithOpenOpp.contains(acc.Id)) {
            acc.addError(
                'Cannot delete this Account because it has Open Opportunities.'
            );
        }
    }
}

trigger AutoCreateContact on Account (after insert) {
    List<Contact> contactList = new List<Contact>();
    for (Account acc : Trigger.new) {
        Contact con = new Contact();
        con.FirstName = 'Default';
        con.LastName = 'Contact';
        con.AccountId = acc.Id;
        contactList.add(con);
    }
    if (!contactList.isEmpty()) {
        insert contactList;
    }
}

public class UpdateContactMailingCity
    implements Queueable {
    private Map<Id, String> accountCityMap;
    public UpdateContactMailingCity(
        Map<Id, String> accountCityMap
    ) {
        this.accountCityMap = accountCityMap;
    }
    public void execute(QueueableContext context) {
        List<Contact> contactList = [
            SELECT Id, AccountId, MailingCity
            FROM Contact
            WHERE AccountId IN :accountCityMap.keySet()
        ];
        for (Contact con : contactList) {
            con.MailingCity =
                accountCityMap.get(con.AccountId);
        }
        if (!contactList.isEmpty()) {
            update contactList;
        }
    }
}

trigger AccountBillingCityTrigger on Account (after update) {
    Map<Id, String> accountCityMap =
        new Map<Id, String>();
    for (Account acc : Trigger.new) {
        Account oldAcc = Trigger.oldMap.get(acc.Id);
        if (acc.BillingCity != oldAcc.BillingCity) {

            accountCityMap.put(
                acc.Id,
                acc.BillingCity
            );
        }
    }
    if (!accountCityMap.isEmpty()) {
        System.enqueueJob(
            new UpdateContactMailingCity(accountCityMap)
        );
    }
}
